// Legacy values intentionally redacted. Runtime production no longer uses Supabase.
const SUPABASE_URL = 'https://YOUR_PROJECT.supabase.co';
const SUPABASE_KEY = 'REDACTED_LEGACY_KEY';

function sbFetchServer(endpoint, method, payload, prefer) {
  const url = `${SUPABASE_URL}/rest/v1/${endpoint}`;
  const options = {
    method: method || 'GET',
    headers: { 'apikey': SUPABASE_KEY, 'Authorization': 'Bearer ' + SUPABASE_KEY, 'Content-Type': 'application/json' },
    muteHttpExceptions: true
  };
  if (prefer) options.headers['Prefer'] = prefer;
  if (payload) options.payload = JSON.stringify(payload);
  
  const response = UrlFetchApp.fetch(url, options);
  const text = response.getContentText();
  const code = response.getResponseCode();
  if (code >= 300) throw new Error(`Supabase error (${code}): ${text}`);
  return text ? JSON.parse(text) : null;
}

function verifyAdmin(session) {
  if (!session || String(session.role || '').toLowerCase().trim() !== 'admin') {
    throw new Error('AKSES DITOLAK: Fitur ini khusus Administrator.');
  }
}

function verifyPengelola(session) {
  if (!session || !['admin', 'guru'].includes(String(session.role || '').toLowerCase().trim())) {
    throw new Error('AKSES DITOLAK: Sesi tidak sah.');
  }
}

function loginSiswaAPI(nomor_ujian, pin) {
  try {
    const rows = sbFetchServer(`siswa?select=*&nomor_ujian=eq.${encodeURIComponent(nomor_ujian.trim())}&limit=1`, 'GET');
    if (!rows || rows.length === 0) return { success: false, message: 'Nomor ujian tidak terdaftar.' };
    
    const siswa = rows[0];
    if (siswa.status_aktif === false) return { success: false, message: 'Akun dinonaktifkan Admin.' };
    if (String(siswa.pin).trim() !== String(pin).trim()) return { success: false, message: 'PIN Ujian salah.' };

    const ujianRows = sbFetchServer(`ujian?select=*&tingkat=eq.${encodeURIComponent(siswa.tingkat)}&status_aktif=eq.true`, 'GET') || [];
    const hasilRows = sbFetchServer(`hasil_ujian?select=ujian_id,status&siswa_id=eq.${siswa.id}`, 'GET') || [];

    let listJadwal = [];
    for (const u of ujianRows) {
      if (u.kelas_target && u.kelas_target.trim() !== '') {
        const allowedKelas = u.kelas_target.split(',').map(k => k.trim().toLowerCase());
        if (!allowedKelas.includes(String(siswa.kelas).trim().toLowerCase())) continue;
      }
      
      const hasil = hasilRows.find(h => Number(h.ujian_id) === Number(u.id));
      let statusPengerjaan = 'belum'; 
      if (hasil) statusPengerjaan = String(hasil.status).toLowerCase();
      
      if (siswa.ujian_status === 'dihentikan' && statusPengerjaan === 'dihentikan') {
        statusPengerjaan = 'terblokir';
      }
      
      let infoJadwal = u.tanggal_mulai || u.tanggal_ujian || '-';
      if (u.jam_mulai) infoJadwal += ` (${u.jam_mulai}${u.jam_selesai ? ' - ' + u.jam_selesai : ''})`;

      listJadwal.push({
        id: u.id, 
        nama_ujian: `${u.nama_ujian} (Sesi ${u.sesi || 1})`, 
        durasi_menit: u.durasi_menit, 
        sesi: u.sesi || 1,
        tanggal_ujian: infoJadwal,
        tahun_ajaran: u.tahun_ajaran || '-', 
        semester: u.semester || '-', 
        status_pengerjaan: statusPengerjaan
      });
    }
    return { success: true, siswa: siswa, jadwal: listJadwal };
  } catch (err) { return { success: false, message: err.message }; }
}

function loginPenggunaAPI(username, password) {
  try {
    const cleanUser = String(username || '').trim();
    const cleanPass = String(password || '').trim();
    
    const rows = sbFetchServer(`akun_pengguna?select=*&username=eq.${encodeURIComponent(cleanUser)}&limit=1`, 'GET');
    if (!rows || rows.length === 0) return { success: false, message: `Username "${cleanUser}" tidak ditemukan.` };
    
    const user = rows[0];
    if (user.status_aktif === false || user.status_aktif === 'false') return { success: false, message: 'Akun dinonaktifkan.' };
    
    const dbPass = String(user.password_hash || user.password || user.kata_sandi || user.pass || '').trim();
    if (dbPass !== cleanPass) return { success: false, message: 'Password salah.' };
    
    try { sbFetchServer(`akun_pengguna?id=eq.${user.id}`, 'PATCH', { terakhir_login: new Date().toISOString() }); } catch (e) {}
    
    const cleanRole = String(user.role || 'guru').toLowerCase().trim();
    return { success: true, role: cleanRole, userId: user.id, id: user.id, nama: user.nama_lengkap || user.username };
  } catch (err) { return { success: false, message: 'Error: ' + err.message }; }
}

function simpanJawabanServer(data) {
  try {
    const { siswaId, ujianId, soalId, jawaban, nomorUjian, namaSiswa, kelas, nomorSoal } = data;
    const ext = sbFetchServer(`jawaban_siswa?select=id&siswa_id=eq.${siswaId}&ujian_id=eq.${ujianId}&soal_id=eq.${soalId}&limit=1`, 'GET');
    
    const payload = {
      siswa_id: siswaId, ujian_id: ujianId, soal_id: soalId, jawaban: jawaban,
      nomor_ujian: nomorUjian || '', nama_siswa: namaSiswa || '', kelas: kelas || '',
      nomor_soal: nomorSoal || 1, waktu_jawab: new Date().toISOString()
    };

    if (ext && ext.length > 0) {
      sbFetchServer(`jawaban_siswa?id=eq.${ext[0].id}`, 'PATCH', { jawaban: jawaban, waktu_jawab: new Date().toISOString() });
    } else {
      sbFetchServer(`jawaban_siswa`, 'POST', payload);
    }
    return { success: true };
  } catch (e) { return { success: false, message: e.message }; }
}

function catatPelanggaranServer(siswaId, ujianId) {
  try {
    const logs = sbFetchServer(`log_pelanggaran?select=id&siswa_id=eq.${siswaId}&ujian_id=eq.${ujianId}`, 'GET') || [];
    const jml = logs.length + 1;
    sbFetchServer(`log_pelanggaran`, 'POST', {
      siswa_id: siswaId, ujian_id: ujianId, jenis_pelanggaran: 'MINIMIZE_TAB', jumlah_pelanggaran: jml, keterangan: 'Keluar aplikasi/tab CBT'
    });
    if (jml >= 3) {
      sbFetchServer(`hasil_ujian?siswa_id=eq.${siswaId}&ujian_id=eq.${ujianId}&status=eq.berlangsung`, 'PATCH', { status: 'dihentikan' });
      sbFetchServer(`siswa?id=eq.${siswaId}`, 'PATCH', { ujian_status: 'dihentikan' });
      return { success: true, dihentikan: true, jumlah: jml };
    }
    return { success: true, dihentikan: false, jumlah: jml };
  } catch (e) { return { success: false, message: e.message }; }
}

function submitUjian(data) {
  try {
    const { siswa_id: siswaId, ujian_id: ujianId } = data;
    const soalList = sbFetchServer(`soal?select=id,jawaban_benar,poin&ujian_id=eq.${ujianId}&status_aktif=eq.true`, 'GET') || [];
    const jawabanList = sbFetchServer(`jawaban_siswa?select=soal_id,jawaban&siswa_id=eq.${siswaId}&ujian_id=eq.${ujianId}`, 'GET') || [];
    
    const jawabanMap = {};
    jawabanList.forEach(j => { jawabanMap[j.soal_id] = j.jawaban; });
    
    let jumlahBenar = 0, jumlahSalah = 0, jumlahKosong = 0, totalPoinDiperoleh = 0, totalPoinMaksimum = 0;
    
    soalList.forEach(s => {
      const p = Number(s.poin || 1);
      totalPoinMaksimum += p;
      const jawabSiswa = jawabanMap[s.id];
      
      if (!jawabSiswa) {
        jumlahKosong++;
      } else if (String(jawabSiswa).trim().toUpperCase() === String(s.jawaban_benar).trim().toUpperCase()) {
        jumlahBenar++;
        totalPoinDiperoleh += p;
      } else {
        jumlahSalah++;
      }
    });
    
    const nilai = totalPoinMaksimum > 0 ? Number(((totalPoinDiperoleh / totalPoinMaksimum) * 100).toFixed(2)) : 0;
    
    const hasilRows = sbFetchServer(`hasil_ujian?select=id&siswa_id=eq.${siswaId}&ujian_id=eq.${ujianId}&order=id.desc&limit=1`, 'GET');
    const updateData = {
      jumlah_soal: soalList.length, jumlah_benar: jumlahBenar, jumlah_salah: jumlahSalah,
      jumlah_kosong: jumlahKosong, total_poin: totalPoinDiperoleh, nilai: nilai,
      waktu_selesai: new Date().toISOString(), status: 'selesai', updated_at: new Date().toISOString()
    };
    
    if (hasilRows && hasilRows.length > 0) {
      sbFetchServer(`hasil_ujian?id=eq.${hasilRows[0].id}`, 'PATCH', updateData);
    } else {
      updateData.siswa_id = siswaId; updateData.ujian_id = ujianId; updateData.waktu_mulai = new Date().toISOString();
      sbFetchServer('hasil_ujian', 'POST', updateData);
    }
    
    sbFetchServer(`siswa?id=eq.${siswaId}`, 'PATCH', { ujian_status: 'selesai', ujian_aktif_id: null });
    return { success: true, hasil: { nilai, jumlah_benar: jumlahBenar, jumlah_salah: jumlahSalah, jumlah_kosong: jumlahKosong }};
  } catch (err) { return { success: false, message: 'Gagal memproses nilai: ' + err.message }; }
}

function importSiswaBulk(session, rowsArray) {
  verifyAdmin(session);
  try {
    const formatted = rowsArray.filter(r => r.nomor_ujian && r.nama).map(r => {
      let pinSiswa = r.pin && String(r.pin).trim() !== '' 
        ? String(r.pin).trim() 
        : String(Math.floor(1000 + Math.random() * 9000));

      return {
        nomor_ujian: String(r.nomor_ujian).trim(),
        nama: String(r.nama).trim(),
        kelas: String(r.kelas || '').trim(),
        tingkat: String(r.tingkat || '').trim(),
        pin: pinSiswa,
        tahun_ajaran: String(r.tahun_ajaran || '2025/2026').trim(),
        status_aktif: true
      };
    });

    if (formatted.length === 0) return { success: false, message: 'File tidak berisi data siswa yang valid.' };

    const chunkSize = 200;
    let insertedCount = 0;
    for (let i = 0; i < formatted.length; i += chunkSize) {
      const chunk = formatted.slice(i, i + chunkSize);
      sbFetchServer('siswa', 'POST', chunk, 'return=minimal');
      insertedCount += chunk.length;
    }

    return { success: true, message: `Berhasil mengimport ${insertedCount} data siswa dengan PIN otomatis unik!` };
  } catch (err) { return { success: false, message: err.message }; }
}

function prosesKenaikanKelasAdmin(session) {
  verifyAdmin(session);
  try {
    const siswaList = sbFetchServer('siswa?select=*', 'GET') || [];
    let countXI = 0, countXII = 0, countHapusXII = 0;
    let dataSiswaXIIArsip = [];
    
    for (const s of siswaList) {
      const t = String(s.tingkat || '').toUpperCase().trim();
      if (t === 'XI') {
        sbFetchServer(`siswa?id=eq.${s.id}`, 'PATCH', { tingkat: 'XII' });
        countXII++;
      } else if (t === 'X') {
        sbFetchServer(`siswa?id=eq.${s.id}`, 'PATCH', { tingkat: 'XI' });
        countXI++;
      } else if (t === 'XII') {
        dataSiswaXIIArsip.push({
          nomor_ujian: s.nomor_ujian,
          nama: s.nama,
          kelas: s.kelas,
          tingkat: s.tingkat,
          pin: s.pin,
          tahun_ajaran: s.tahun_ajaran
        });
        sbFetchServer(`siswa?id=eq.${s.id}`, 'DELETE');
        countHapusXII++;
      }
    }
    return { 
      success: true, 
      dataXII: dataSiswaXIIArsip,
      message: `Kenaikan kelas sukses! ${countXI} siswa X naik ke XI, ${countXII} siswa XI naik ke XII, dan ${countHapusXII} siswa XII (tamat) telah diarsipkan/dibersihkan.` 
    };
  } catch (err) { return { success: false, message: 'Gagal proses kenaikan kelas: ' + err.message }; }
}

    const siswaList = sbFetchServer(`siswa?select=id&tingkat=eq.${cleanTingkat}`, 'GET') || [];

function importSoalBulk(session, rowsArray) {
  verifyAdmin(session);
  try {
    const ujianRows = sbFetchServer('ujian?select=id,nama_ujian', 'GET') || [];
    const ujianMap = {};
    ujianRows.forEach(u => {
      if (u.nama_ujian) ujianMap[String(u.nama_ujian).trim().toLowerCase()] = u.id;
    });

    const formatted = rowsArray.filter(r => (r.nama_ujian || r.ujian_id) && r.pertanyaan).map(r => {
      let ujianId = Number(r.ujian_id);
      if (!ujianId && r.nama_ujian) {

        ujian_id: ujianId,

        opsi_a: String(r.opsi_a || ''),
        opsi_b: String(r.opsi_b || ''),

        opsi_d: String(r.opsi_d || ''),
        opsi_e: String(r.opsi_e || ''),
        jawaban_benar: String(r.jawaban_benar || 'A').toUpperCase(),
        poin: Number(r.poin || 1),
        status_aktif: true
      };
    }).filter(Boolean);

    if (formatted.length === 0) return { success: false, message: 'Gagal import: Pastikan kolom nama_ujian di Excel sama persis dengan Nama Ujian di menu Kelola Ujian.' };

    const chunkSize = 200;
    let insertedCount = 0;
    for (let i = 0; i < formatted.length; i += chunkSize) {
      const chunk = formatted.slice(i, i + chunkSize);
      sbFetchServer('soal', 'POST', chunk, 'return=minimal');
      insertedCount += chunk.length;
    }

    return { success: true, message: `Berhasil mengimport ${insertedCount} soal ke database.` };
  } catch (err) { return { success: false, message: err.message }; }
}

function getAdminGuruUjianList(session) {
  verifyAdmin(session);
  try {
    const relasi = sbFetchServer('guru_ujian?select=*&order=id.desc', 'GET') || [];
    const akun = sbFetchServer('akun_pengguna?select=id,username,nama_lengkap&role=eq.guru', 'GET') || [];
    const ujian = sbFetchServer('ujian?select=id,nama_ujian,tingkat,sesi', 'GET') || [];
    
    const akunMap = {}; akun.forEach(a => akunMap[a.id] = a);
    const ujianMap = {}; ujian.forEach(u => ujianMap[u.id] = u);
    
    const merged = relasi.map(r => ({
      id: r.id,
      guru_id: r.guru_id,
      ujian_id: r.ujian_id,
      nama_guru: akunMap[r.guru_id]?.nama_lengkap || akunMap[r.guru_id]?.username || 'Guru ID ' + r.guru_id,
      nama_ujian: (ujianMap[r.ujian_id]?.nama_ujian || 'Ujian ID ' + r.ujian_id) + ` (Sesi ${ujianMap[r.ujian_id]?.sesi || 1})`,
      tingkat: ujianMap[r.ujian_id]?.tingkat || '-'

function getGuruExamResults(session) {
  verifyPengelola(session);
  try {
    const userId = session.userId || session.id;
    const cleanRole = String(session.role).toLowerCase().trim();
    let ujianIds = [];
    
    if (cleanRole === 'admin') {
      const allUjian = sbFetchServer('ujian?select=id', 'GET') || [];
      ujianIds = allUjian.map(u => u.id);
    } else {
      const relasi = sbFetchServer(`guru_ujian?select=ujian_id&guru_id=eq.${userId}`, 'GET') || [];
      ujianIds = relasi.map(r => r.ujian_id);
    }
    
    if (ujianIds.length === 0) return { success: true, ujianList: [], hasilList: [], pelanggaranList: [] };
    
    const ujianListRaw = sbFetchServer(`ujian?select=*&id=in.(${ujianIds.join(',')})`, 'GET') || [];
    const ujianList = ujianListRaw.map(u => {
      let jStr = u.tanggal_mulai || 'Tanggal TBD';
      if (u.jam_mulai) jStr += ` (${u.jam_mulai}${u.jam_selesai ? ' - ' + u.jam_selesai : ''})`;
      return {
        ...u,
        nama_ujian: `${u.nama_ujian} (Sesi ${u.sesi || 1} - ${jStr})`
      };
    });

    const hasilListAll = sbFetchServer(`hasil_ujian?select=*&ujian_id=in.(${ujianIds.join(',')})&order=nilai.desc`, 'GET') || [];
    const pelanggaranListAll = sbFetchServer(`log_pelanggaran?select=*&ujian_id=in.(${ujianIds.join(',')})`, 'GET') || [];
    
    const siswaRows = sbFetchServer('siswa?select=id,kelas,nomor_ujian,nama', 'GET') || [];
    const siswaMap = {}; 
    siswaRows.forEach(s => { siswaMap[s.id] = s; });
    
    const hasilList = hasilListAll.map(h => {
      const s = siswaMap[h.siswa_id] || {};
      return {
        ...h,
        nomor_ujian: h.nomor_ujian || s.nomor_ujian || '-',
        nama_siswa: h.nama_siswa || s.nama || '-',
        kelas: h.kelas || s.kelas || '-'
      };
    }).filter(h => h.siswa_id != null);
    
    const pelanggaranList = pelanggaranListAll.map(p => {
      const s = siswaMap[p.siswa_id] || {};
      return {
        ...p,
        nomor_ujian: p.nomor_ujian || s.nomor_ujian || '-',
        nama_siswa: p.nama_siswa || s.nama || '-',
        kelas: p.kelas || s.kelas || '-'
      };
    }).filter(p => p.siswa_id != null);
    
    return { success: true, ujianList, hasilList, pelanggaranList };
  } catch (err) { 
    return { success: false, message: err.message }; 
  }
}

function updatePasswordGuru(session, oldPass, newPass) {
  verifyPengelola(session);
  try {
    const userId = session.userId || session.id;
    const rows = sbFetchServer(`akun_pengguna?select=*&id=eq.${userId}&limit=1`, 'GET');
    if (!rows || rows.length === 0) return { success: false, message: "Akun tidak ditemukan." };
    
    const user = rows[0];
    const dbPass = String(user.password_hash || user.password || '').trim();
    if (dbPass !== String(oldPass).trim()) {
      return { success: false, message: "Password lama salah." };
    }
    
    sbFetchServer(`akun_pengguna?id=eq.${userId}`, 'PATCH', { password_hash: String(newPass).trim() });
    return { success: true, message: "Password berhasil diubah!" };
  } catch (err) {
    return { success: false, message: err.message };
  }
}
